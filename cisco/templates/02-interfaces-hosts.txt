! ===== 02 INTERFACE IPs + HOSTS =====
! --- Router interface ---
configure terminal
 interface <g0/0 or fa0/0>
  description <TO WHAT>
  ! e.g. 64.103.100.101 255.255.255.0
  ip address <IP> <MASK>
  ! interfaces start OFF on routers
  no shutdown
  exit
!
! --- Layer 2 switch management IP (2960) ---
 interface vlan 1
  ip address <IP> <MASK>
  no shutdown
  exit
 ip default-gateway <ROUTER_IP_ON_THAT_LAN>
!
! --- Shut unused switch ports (if asked) ---
 interface range fa0/10 - 24
  shutdown
  exit
end
write memory
! VERIFY: show ip interface brief   -> Status up, Protocol up
