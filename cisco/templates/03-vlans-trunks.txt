! ===== 03 VLANs, ACCESS PORTS, TRUNKS, INTER-VLAN ROUTING =====
configure terminal
 vlan <10>
  name <STUDENTS>
  exit
 vlan <20>
  name <STAFF>
  exit
!
! --- Access port (one VLAN, for a PC) ---
 interface fa0/<2>
  switchport mode access
  switchport access vlan <10>
  ! only on ports to end devices
  spanning-tree portfast
  ! only if asked
  spanning-tree bpduguard enable
  exit
!
! --- Trunk (switch-to-switch / switch-to-router) ---
 interface g0/1
  ! ONLY on 3560/3650 L3 switches; 2960 rejects it (that's fine)
  switchport trunk encapsulation dot1q
  switchport mode trunk
  ! ONE line, comma list. Three separate lines overwrite each other!
  switchport trunk allowed vlan 1,10,20
  ! only if asked
  switchport trunk native vlan <99>
  exit
end
write memory
!
! ===== Inter-VLAN routing option A: Layer 3 switch (3560/3650) =====
configure terminal
 ! without this the L3 switch will NOT route
 ip routing
 interface vlan 10
  ip address <10.0.10.1> <255.255.255.0>
  no shutdown
 interface vlan 20
  ip address <10.0.20.1> <255.255.255.0>
  no shutdown
 ! routed uplink port
 interface g1/0/1
  no switchport
  ip address <IP> <MASK>
  no shutdown
end
!
! ===== Option B: router-on-a-stick =====
configure terminal
 interface g0/0
  no shutdown
 interface g0/0.10
  encapsulation dot1Q 10
  ip address <10.0.10.1> <255.255.255.0>
 interface g0/0.20
  encapsulation dot1Q 20
  ip address <10.0.20.1> <255.255.255.0>
end
! VERIFY: show vlan brief ; show interfaces trunk ; show ip route (C routes for each VLAN)
